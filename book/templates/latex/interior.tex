% #1 label, #2 chapter number or empty, #3 title, #4 running head, #5 id, #6 listed (1/0)
\newcommand\aixpchapter[6]{%
  \gdef\aixp@chapterlabel{#1}%
  \chapter*{#3}%
  \gdef\aixp@chapterlabel{}%
  \ifnum#6=1
    \if\relax\detokenize{#2}\relax
      \addcontentsline{toc}{chapter}{#3}%
    \else
      \addcontentsline{toc}{chapter}{\protect\numberline{#2}#3}%
    \fi
  \fi
  \chaptermark{#4}\label{#5}}
